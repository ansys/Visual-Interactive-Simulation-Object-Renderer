```
class SceneLock(Protocol):
    """The two lock operations the window uses."""
    def acquire(self, blocking: bool = True) -> bool: """Acquire; non-blocking returns False when held elsewhere."""
    def release(self) -> None: """Release one level."""

class VisorRcaWindow:
    """trame-rca area over a ServerRenderer that touches VTK only while holding the scene lock."""
    def __init__(self, capture: RemoteControlledAreaProtocol, renderer: ServerRenderer, vtk_lock: SceneLock, reschedule: Callable[[], None]) -> None: """All four required."""
    @property
    def img_cols_rows(self) -> tuple[NDArray, int, int]: """Render and capture under the lock; when another thread holds it, return the last frame and reschedule."""
    def process_resize_event(self, width: int, height: int) -> None: """Record the size; the last size recorded is applied under the lock at the next frame."""
    def process_interaction_event(self, event: dict) -> None: """Queue the event; every queued event is forwarded in order under the lock at the next frame."""
    def _apply_deferred(self) -> None: """Apply the last recorded size once, then forward every queued event in arrival order; caller holds the lock."""
    def _placeholder(self) -> tuple[NDArray, int, int]: """A frame of the renderer's background at the last recorded size, (300, 300) before any."""
```
* The behaviour is Section 4 steps 1–3, unchanged.
* The queue is a plain list and the recorded size a plain field. All three entry points run on the loop thread: trame-rca's RPC handlers run as trigger handlers do (item 1), and the scheduler calls img_cols_rows there [src] [inf for the RPC handlers].

`app/trame/server_loop_bridge.py` (new)
```
class ServerLoopBridge(FrameRequester):
    """Carries frame requests and state writes from any thread onto the trame event loop."""
    RETRY_DELAY_S: float = 0.1
    def __init__(self, state: State) -> None: """Unbound until the server is ready."""
    def bind(self, view_handler: RcaViewAdapter) -> None: """Start delivering on the running loop; called once on the loop thread, then requests one frame."""
    def request_frame(self) -> None: """Schedule a render on the loop thread; dropped before bind."""
    def retry_frame(self) -> None: """Ask for one frame after RETRY_DELAY_S unless a retry is already pending; loop thread only; dropped before bind."""
    def write_state(self, key: str, value: object) -> None: """Set one state key and flush it on the loop thread; written directly before bind."""
    def _fire_retry(self) -> None: """Clear the pending mark, then schedule a render."""
    def _write_and_flush(self, key: str, value: object) -> None: """Set one key and flush; runs on the loop thread."""
```
* request_frame → loop.call_soon_threadsafe(handler.schedule_render).
* bind reads asyncio.get_running_loop() itself, then calls handler.schedule_render() directly, since it is already on the loop thread.
* retry_frame and _fire_retry run only on the loop thread, so the pending flag needs no lock.
* retry_frame can't be reached before bind in production: S1.6b builds the handler and calls bind in the same synchronous on_server_ready, and the scheduler's tasks start only afterwards [inf]. Before bind it is a guard drop, not a stand-in.
* state and reschedule take no default.
